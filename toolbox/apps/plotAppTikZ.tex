% Draw a slope triangle in a log-log plot.
%  #1: upper-left corner, x-coordinate
%  #2: upper-left corner, y-coordinate
%  #3: positive slope magnitude (triangle shows slope -#3)
%  #4: horizontal size (in decades, e.g. 0.5 means ×10^{0.5})
%  #5: tikz style, e.g. thick
% Usage: \\logLogSlopeTriangleTop{x0}{y0}{s}{L}{style}
% Example: \\logLogSlopeTriangleTop{1e-1}{1e-1}{2}{1}{thick}
\newcommand{\logLogSlopeTriangleTop}[5]{%
  \pgfplotsextra{%
    \pgfkeys{/pgf/fpu=true, pgf/fpu/output format=fixed}
    \pgfmathsetmacro{\xA}{#1}
    \pgfmathsetmacro{\yA}{#2}
    \pgfmathsetmacro{\xB}{\xA * 10^(#4)}
    \pgfmathsetmacro{\yB}{\yA}
    \pgfmathsetmacro{\xC}{\xB}
    \pgfmathsetmacro{\yC}{\yA / 10^(#3 * #4)}
    \pgfkeys{/pgf/fpu=false}
    \draw[#5]
      (\xA,\yA) -- node[pos=0.5,above=-2pt]{\small 1}
      (\xB,\yB) -- node[pos=0.5,right=-2pt]{\small #3}
      (\xC,\yC) -- cycle;
  }
}

\newcommand{\logLogSlopeTriangleBottom}[5]{%
  \pgfplotsextra{%
    \pgfkeys{/pgf/fpu=true, pgf/fpu/output format=fixed}
    \pgfmathsetmacro{\xA}{#1}
    \pgfmathsetmacro{\yA}{#2}
    \pgfmathsetmacro{\xC}{\xA * 10^(#4)}
    \pgfmathsetmacro{\yC}{\yA / 10^(#3 * #4)}
    \pgfmathsetmacro{\xB}{\xA}
    \pgfmathsetmacro{\yB}{\yC}
    \pgfkeys{/pgf/fpu=false}
    \draw[#5]
      (\xA,\yA) -- node[pos=0.5,left=-2pt]{\small #3}
      (\xB,\yB) -- node[pos=0.5,below=-2pt]{\small 1}
      (\xC,\yC) -- cycle;
  }
}
